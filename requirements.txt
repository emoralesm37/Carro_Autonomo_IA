# PASO 1 — Instalar paquetes del sistema (apt):
#   sudo apt update
#   sudo apt install -y \
#     python3-picamera2 \
#     python3-opencv    \
#     python3-gpiozero  \
#     python3-lgpio     \
#     python3-pip
#
# PASO 2 — Instalar paquetes Python (pip):
#   pip install -r requirements.txt
#
# PASO 3 — Habilitar cámara (una sola vez):
#   sudo raspi-config → Interface Options → Camera → Enable
#   (o agregar camera_auto_detect=1 en /boot/firmware/config.txt)
#
# PASO 4 — Hardware PWM para motores (opcional, mejora suavidad):
#   Agregar en /boot/firmware/config.txt:
#     dtoverlay=pwm-2chan,pin=12,func=4,pin2=13,func2=4
#   Luego reiniciar.
#
